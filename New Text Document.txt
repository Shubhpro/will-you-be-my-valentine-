<script>

/* CUSTOMIZE */
const HER_NAME = "Aanya";

const TYPING_MESSAGE =
  "Aanya 💖\n\n" +
  "From our late-night talks and silly fights 😅\n" +
  "To all memories together💕\n\n" +
  "I love you my doll 💕";

/* NAME */
document.getElementById("herName").innerText = HER_NAME;

/* QR */
new QRCode(document.getElementById("qr"), {
  text: window.location.href,
  width: 120,
  height: 120
});

/* MUSIC */
document.body.addEventListener("click", () => {
  document.getElementById("bgMusic").play();
}, { once: true });

/* TYPING */
const typingElement = document.getElementById("typingText");
let index = 0;

function typeText() {

  if (index < TYPING_MESSAGE.length) {

    typingElement.innerHTML +=
      TYPING_MESSAGE[index] === "\n"
        ? "<br>"
        : TYPING_MESSAGE[index];

    index++;

    setTimeout(typeText, 40);
  }
}

window.onload = typeText;

/* MOVE NO */
function moveNo() {

  const btn = document.getElementById("no");

  btn.style.left =
    Math.random() * (window.innerWidth - 120) + "px";

  btn.style.top =
    Math.random() * (window.innerHeight - 60) + "px";
}

/* CONFETTI */
const canvas = document.getElementById("confetti");
const ctx = canvas.getContext("2d");

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

let confetti = [];

function createConfetti() {

  confetti = [];

  for (let i = 0; i < 160; i++) {

    confetti.push({
      x: Math.random() * canvas.width,
      y: Math.random() * canvas.height - canvas.height,
      r: Math.random() * 6 + 4,
      d: Math.random() * 30,
      color: `hsl(${Math.random()*360},100%,50%)`
    });

  }
}

function drawConfetti() {

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  confetti.forEach(c => {

    ctx.beginPath();
    ctx.arc(c.x, c.y, c.r, 0, Math.PI * 2);

    ctx.fillStyle = c.color;
    ctx.fill();
  });

  updateConfetti();
}

function updateConfetti() {

  confetti.forEach(c => {

    c.y += Math.cos(c.d) + 4;

    if (c.y > canvas.height) {
      c.y = -10;
    }

  });
}

/* YES BUTTON */
function sayYes() {

  createConfetti();
  setInterval(drawConfetti, 20);

  document.body.innerHTML = `

    align-items:center;

    <h1>GREAT CHOICE 💕🎉</h1>

    <p style="font-size:22px;">
      Happy Valentine’s Day ${HER_NAME} 💖<br>
      I’m so lucky to have you 🥰
    </p>

  </div>

  `;
}
